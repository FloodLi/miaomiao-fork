cmake_minimum_required(VERSION 3.16)
project(epd_fault_tests C)
enable_testing()
add_executable(epd_fault_tests test_epd.c)
target_include_directories(epd_fault_tests PRIVATE include ../../main)
target_compile_features(epd_fault_tests PRIVATE c_std_11)
target_compile_options(epd_fault_tests PRIVATE -Wall -Wextra -Werror=return-type -Werror=implicit-function-declaration -Wno-unused-parameter -Wno-missing-field-initializers)
option(EPD_SANITIZE "Enable address and undefined behavior sanitizers" ON)
if(EPD_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(epd_fault_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
    target_link_options(epd_fault_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME epd_faults COMMAND epd_fault_tests)
